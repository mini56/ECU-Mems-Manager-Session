cmake_minimum_required(VERSION 3.16)
project(mems16na_analysis_test LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(mems16na_analysis_test test.cpp "${ROOT}/i18n.cpp" "${ROOT}/i18n.h")
target_include_directories(mems16na_analysis_test PRIVATE "${ROOT}")
target_link_libraries(mems16na_analysis_test PRIVATE Qt5::Widgets)
if(MSVC)
    target_compile_options(mems16na_analysis_test PRIVATE /utf-8)
endif()
add_custom_command(TARGET mems16na_analysis_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${ROOT}/translations" "$<TARGET_FILE_DIR:mems16na_analysis_test>/translations")
