cmake_minimum_required(VERSION 3.16)
project(mems16na_diagnostic_test LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(mems16na_diagnostic_test test.cpp test_resources.qrc "${ROOT}/diagnosticpanel.cpp" "${ROOT}/diagnosticpanel.h" "${ROOT}/summarytab.cpp" "${ROOT}/summarytab.h" "${ROOT}/i18n.cpp" "${ROOT}/i18n.h")
target_include_directories(mems16na_diagnostic_test PRIVATE "${ROOT}")
target_link_libraries(mems16na_diagnostic_test PRIVATE Qt5::Widgets)
if(WIN32)
  target_compile_definitions(mems16na_diagnostic_test PRIVATE WIN32)
endif()
if(MSVC)
  target_compile_options(mems16na_diagnostic_test PRIVATE /utf-8)
endif()
add_custom_command(TARGET mems16na_diagnostic_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${ROOT}/translations" "$<TARGET_FILE_DIR:mems16na_diagnostic_test>/translations")
